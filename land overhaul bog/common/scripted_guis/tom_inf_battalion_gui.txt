	tom_inf_battalion_editor_terrain_gui = {
		context_type = player_context
		window_name = "tom_inf_bat_terrain_window"
		parent_window_name = "tom_inf_battalion_designer_window"
		
		triggers = {
			terrain_movement_visible = {
				if = {
					limit = {
						check_variable = { tom_inf_bat_popup_terrain_movement_array^i = 0 }
					}
					set_temp_variable = { movement_frame = 3 }
				}
				else_if = {
					limit = {
						check_variable = { tom_inf_bat_popup_terrain_movement_array^i > 0 }
					}
					set_temp_variable = { movement_frame = 1 }
				}
				else = {
					set_temp_variable = { movement_frame = 2 }
				}
			}
			
			terrain_attack_visible = {
				if = {
					limit = {
						check_variable = { tom_inf_bat_popup_terrain_attack_array^i = 0 }
					}
					set_temp_variable = { attack_frame = 3 }
				}
				else_if = {
					limit = {
						check_variable = { tom_inf_bat_popup_terrain_attack_array^i > 0 }
					}
					set_temp_variable = { attack_frame = 1 }
				}
				else = {
					set_temp_variable = { attack_frame = 2 }
				}
			}
			
			terrain_defense_visible = {
				if = {
					limit = {
						check_variable = { tom_inf_bat_popup_terrain_defense_array^i = 0 }
					}
					set_temp_variable = { defense_frame = 3 }
				}
				else_if = {
					limit = {
						check_variable = { tom_inf_bat_popup_terrain_defense_array^i > 0 }
					}
					set_temp_variable = { defense_frame = 1 }
				}
				else = {
					set_temp_variable = { defense_frame = 2 }
				}
			}
		}
		
		properties = {
			terrain_icon = { image = [tom_inf_bat_terrain_icon_loc] }
			terrain_movement = { frame = movement_frame }
			terrain_attack = { frame = attack_frame }
			terrain_defense = { frame = defense_frame }
		}
		
		dynamic_lists = {
			terrain_stats_grid = {
				array = tom_inf_bat_popup_terrain_bonus_array
				entry_container = tom_inf_bat_terrain_entry
			}
		}
