tom_inf_support_equipment_production = {
	allowed = { always = yes }
	enable = {
		OR = {
			arms_factory > 19
			is_major = yes
		}
		has_tech = support_weapons
	}
	abort_when_not_enabled = yes
	
	ai_strategy = {
		type = equipment_production_min_factories_archetype
		id = lmg_chassis
		value = 1
	}
	ai_strategy = {
		type = equipment_production_min_factories_archetype
		id = mmg_chassis
		value = 1
	}
	ai_strategy = {
		type = equipment_production_min_factories_archetype
		id = hmg_chassis
		value = 1
	}
	ai_strategy = {
		type = equipment_production_min_factories_archetype
		id = mortar_chassis
		value = 1
	}
}

tom_inf_infantry_equipment = {
	allowed = { always = yes }
	enable = {
		stockpile_ratio = {
			archetype = infantry_equipment
			ratio < 1.0
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = equipment_variant_production_factor
		id = infantry_equipment
		value = 15
	}
}
