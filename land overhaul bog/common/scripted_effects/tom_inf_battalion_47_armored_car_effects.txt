tom_inf_bat_save_designer_armored_car_effect = {
	if = {
		limit = {
			OR = {
				tom_inf_save_button_increase_armored_car_trigger = yes
				tom_inf_save_button_decrease_armored_car_trigger = yes
			}
		}
		tom_inf_bat_add_effect_armored_car = yes
		tom_inf_bat_remove_effect_armored_car = yes
		
		if = {
			limit = {
				check_variable = { tom_inf_bat_popup_platoon_count_org > tom_inf_bat_armored_car_platoon_count_org }
			}
			tom_inf_bat_increase_platoon_org_armored_car = yes
		}
		else_if = {
			limit = {
				check_variable = { tom_inf_bat_popup_platoon_count_org < tom_inf_bat_armored_car_platoon_count_org }
			}
			tom_inf_bat_decrease_platoon_org_armored_car = yes
		}
		
		clear_array = tom_inf_bat_armored_car_base_stats_array
		clear_array = tom_inf_bat_armored_car_combat_stats_array
		clear_array = tom_inf_bat_armored_car_equipment_stats_array_1
		clear_array = tom_inf_bat_armored_car_equipment_stats_array_2_type
		clear_array = tom_inf_bat_armored_car_equipment_stats_array_2_count
		
		clear_array = tom_inf_bat_armored_car_terrain_bonus_array
		clear_array = tom_inf_bat_armored_car_terrain_movement_array
		clear_array = tom_inf_bat_armored_car_terrain_attack_array
		clear_array = tom_inf_bat_armored_car_terrain_defense_array
		
		for_each_loop = {
			array = tom_inf_bat_popup_base_stats_array
			value = temp_v
			add_to_array = { tom_inf_bat_armored_car_base_stats_array = temp_v }
		}
		for_each_loop = {
			array = tom_inf_bat_popup_combat_stats_array
			value = temp_v
			add_to_array = { tom_inf_bat_armored_car_combat_stats_array = temp_v }
		}

			array = tom_inf_bat_popup_equipment_stats_array_1
			value = temp_v
			add_to_array = { tom_inf_bat_armored_car_equipment_stats_array_1 = temp_v }
		}
		for_each_loop = {
			array = tom_inf_bat_popup_equipment_stats_array_2_type
			value = temp_v
			add_to_array = { tom_inf_bat_armored_car_equipment_stats_array_2_type = temp_v }
		}
		for_each_loop = {
			array = tom_inf_bat_popup_equipment_stats_array_2_count
			value = temp_v
			add_to_array = { tom_inf_bat_armored_car_equipment_stats_array_2_count = temp_v }
		}

			array = tom_inf_bat_popup_terrain_bonus_array

		}
		for_each_loop = {
			array = tom_inf_bat_popup_terrain_movement_array
			value = temp_v
			add_to_array = { tom_inf_bat_armored_car_terrain_movement_array = temp_v }
		}
		for_each_loop = {
			array = tom_inf_bat_popup_terrain_attack_array
			value = temp_v
			add_to_array = { tom_inf_bat_armored_car_terrain_attack_array = temp_v }
		}
		for_each_loop = {
			array = tom_inf_bat_popup_terrain_defense_array
			value = temp_v
			add_to_array = { tom_inf_bat_armored_car_terrain_defense_array = temp_v }
		}
	}
	
	clear_array = tom_inf_battalion_armored_car_company_array
	clear_array = tom_inf_battalion_armored_car_platoon_array
