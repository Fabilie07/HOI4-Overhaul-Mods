tom_inf_bat_save_designer_modern_tank_effect = {
	if = {
		limit = {
			OR = {
				tom_inf_save_button_increase_modern_tank_trigger = yes
				tom_inf_save_button_decrease_modern_tank_trigger = yes
			}
		}
		tom_inf_bat_add_effect_modern_tank = yes
		tom_inf_bat_remove_effect_modern_tank = yes
		
		if = {
			limit = {
				check_variable = { tom_inf_bat_popup_platoon_count_org > tom_inf_bat_modern_tank_platoon_count_org }
			}
			tom_inf_bat_increase_platoon_org_modern_tank = yes
		}
		else_if = {
			limit = {
				check_variable = { tom_inf_bat_popup_platoon_count_org < tom_inf_bat_modern_tank_platoon_count_org }
			}
			tom_inf_bat_decrease_platoon_org_modern_tank = yes
		}
		
		clear_array = tom_inf_bat_modern_tank_base_stats_array
		clear_array = tom_inf_bat_modern_tank_combat_stats_array
		clear_array = tom_inf_bat_modern_tank_equipment_stats_array_1
		clear_array = tom_inf_bat_modern_tank_equipment_stats_array_2_type
		clear_array = tom_inf_bat_modern_tank_equipment_stats_array_2_count
		
		clear_array = tom_inf_bat_modern_tank_terrain_bonus_array
		clear_array = tom_inf_bat_modern_tank_terrain_movement_array
		clear_array = tom_inf_bat_modern_tank_terrain_attack_array
		clear_array = tom_inf_bat_modern_tank_terrain_defense_array
		
		for_each_loop = {
			array = tom_inf_bat_popup_base_stats_array
			value = temp_v
			add_to_array = { tom_inf_bat_modern_tank_base_stats_array = temp_v }
		}
		for_each_loop = {
			array = tom_inf_bat_popup_combat_stats_array
			value = temp_v
			add_to_array = { tom_inf_bat_modern_tank_combat_stats_array = temp_v }
		}
		for_each_loop = {
			array = tom_inf_bat_popup_equipment_stats_array_1
			value = temp_v
			add_to_array = { tom_inf_bat_modern_tank_equipment_stats_array_1 = temp_v }
		}
		for_each_loop = {
			array = tom_inf_bat_popup_equipment_stats_array_2_type
			value = temp_v
			add_to_array = { tom_inf_bat_modern_tank_equipment_stats_array_2_type = temp_v }
		}
		for_each_loop = {
			array = tom_inf_bat_popup_equipment_stats_array_2_count
			value = temp_v
			add_to_array = { tom_inf_bat_modern_tank_equipment_stats_array_2_count = temp_v }
		}
		
		for_each_loop = {
			array = tom_inf_bat_popup_terrain_bonus_array
			value = temp_v
			add_to_array = { tom_inf_bat_modern_tank_terrain_bonus_array = temp_v }
		}
		for_each_loop = {
			array = tom_inf_bat_popup_terrain_movement_array
			value = temp_v
			add_to_array = { tom_inf_bat_modern_tank_terrain_movement_array = temp_v }
		}
		for_each_loop = {
			array = tom_inf_bat_popup_terrain_attack_array
			value = temp_v
			add_to_array = { tom_inf_bat_modern_tank_terrain_attack_array = temp_v }
		}
		for_each_loop = {
			array = tom_inf_bat_popup_terrain_defense_array
			value = temp_v
			add_to_array = { tom_inf_bat_modern_tank_terrain_defense_array = temp_v }
		}
	}
	
	clear_array = tom_inf_battalion_modern_tank_company_array
	clear_array = tom_inf_battalion_modern_tank_platoon_array
	
	for_each_loop = {
		array = tom_inf_battalion_company_array
		value = temp_v # Avoid any potential name overlap
		add_to_array = { tom_inf_battalion_modern_tank_company_array = temp_v }
	}
	for_each_loop = {
		array = tom_inf_battalion_platoon_array
		value = temp_v
		add_to_array = { tom_inf_battalion_modern_tank_platoon_array = temp_v }
	}

	set_variable = { tom_inf_bat_modern_tank_38_count = tom_inf_bat_popup_38_count }
	
	set_variable = { tom_inf_bat_modern_tank_4600_count = tom_inf_bat_popup_4600_count }
	set_variable = { tom_inf_bat_modern_tank_4010_count = tom_inf_bat_popup_4010_count }
