tom_inf_bat_increase_platoon_org_ranger = {
	while_loop_effect = {
		limit = { check_variable = { tom_inf_bat_ranger_platoon_count_org < tom_inf_bat_popup_platoon_count_org } }
		
		if = {
			limit = { check_variable = { tom_inf_bat_ranger_platoon_count_org > 19 } }
			add_unit_bonus = {
				ranger_battalion = {
					max_organisation = -9
					soft_attack = -0.1
				}
			}
		}
		else_if = {
			limit = { check_variable = { tom_inf_bat_ranger_platoon_count_org > 15 } }
			add_unit_bonus = {
				ranger_battalion = {
					max_organisation = -6
				}
			}
		}
		else_if = {
			limit = { check_variable = { tom_inf_bat_ranger_platoon_count_org > 11 } }
			add_unit_bonus = {
				ranger_battalion = {
					max_organisation = -3
				}
			}
		}
		
		add_to_variable = { tom_inf_bat_ranger_platoon_count_org = 1 }
	}
}
