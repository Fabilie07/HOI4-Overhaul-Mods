tom_inf_bat_add_effect_medium_tank = {
	# Adds necessary stats after saving a larger Battalion.
	
	### PLATOONS
	# Medium Tank
	set_temp_variable = { platoon_difference = tom_inf_bat_medium_tank_4200_count }
	while_loop_effect = {
		limit = { check_variable = { platoon_difference < tom_inf_bat_popup_4200_count } }
		add_unit_bonus = {
			medium_armor = {
				max_organisation = 2
				max_strength = 0.2
				need = {
					medium_tank_chassis = 5
				}
			}
		}

					light_tank_chassis = 5

			}
		}
		
		if = {
			limit = { check_variable = { platoon_difference = 0 } }

				medium_armor = {
					soft_attack = -0.4

tom_inf_bat_increase_platoon_org_medium_tank = {
	while_loop_effect = {
		limit = { check_variable = { tom_inf_bat_medium_tank_platoon_count_org < tom_inf_bat_popup_platoon_count_org } }
		
		if = {
			limit = { check_variable = { tom_inf_bat_medium_tank_platoon_count_org > 19 } }
			add_unit_bonus = {
				medium_armor = {
					max_organisation = -9
					soft_attack = -0.1
				}
			}
		}
		else_if = {
			limit = { check_variable = { tom_inf_bat_medium_tank_platoon_count_org > 15 } }
			add_unit_bonus = {
				medium_armor = {
					max_organisation = -6
				}
			}
		}
		else_if = {
			limit = { check_variable = { tom_inf_bat_medium_tank_platoon_count_org > 11 } }
			add_unit_bonus = {
				medium_armor = {
					max_organisation = -3
				}
			}
		}
		
		add_to_variable = { tom_inf_bat_medium_tank_platoon_count_org = 1 }
	}
}
