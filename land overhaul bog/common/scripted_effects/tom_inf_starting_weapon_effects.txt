		create_equipment_variant = {

		}
		create_equipment_variant = {
			type = sniper_chassis_0
			mark_older_equipment_obsolete = yes
			modules = {
				firing_mode_slot = inf_bolt_action
				magazine_type_slot = inf_clip_05
				backpack_type_slot = inf_backpack_3
				sights_slot = inf_sights_sniper
				
				barrel_slot = inf_barrel_rifle
				cartridge_slot = inf_cartridge_rifle_75
				sidearm_weapon_slot = inf_sidearm_pistol
			}
		}
	}
	
	create_equipment_variant = {
		type = lmg_chassis_0
		mark_older_equipment_obsolete = yes
		modules = {
			mg_rpm_slot = mg_rpm_0500
		}
	}
	
	if = {
		limit = {
			has_tech = support_weapons
		}
		create_equipment_variant = {
			type = mmg_chassis_0

			type = hmg_chassis_0
			mark_older_equipment_obsolete = yes
			modules = {
				mg_rpm_slot = mg_rpm_0500
			}

}
