tom_inf_bat_save_designer_artillery_effect = {
	if = {
		limit = {
			OR = {
				tom_inf_save_button_increase_artillery_trigger = yes
				tom_inf_save_button_decrease_artillery_trigger = yes
			}
		}
		tom_inf_bat_add_effect_artillery = yes
		tom_inf_bat_remove_effect_artillery = yes
		
		if = {
			limit = {
				check_variable = { tom_inf_bat_popup_platoon_count_org > tom_inf_bat_artillery_platoon_count_org }
			}
			tom_inf_bat_increase_platoon_org_artillery = yes
		}
		else_if = {
			limit = {
				check_variable = { tom_inf_bat_popup_platoon_count_org < tom_inf_bat_artillery_platoon_count_org }
			}
			tom_inf_bat_decrease_platoon_org_artillery = yes
		}
		
		clear_array = tom_inf_bat_artillery_base_stats_array
		clear_array = tom_inf_bat_artillery_combat_stats_array
		clear_array = tom_inf_bat_artillery_equipment_stats_array_1
		clear_array = tom_inf_bat_artillery_equipment_stats_array_2_type
		clear_array = tom_inf_bat_artillery_equipment_stats_array_2_count
		
		clear_array = tom_inf_bat_artillery_terrain_bonus_array
		clear_array = tom_inf_bat_artillery_terrain_movement_array
		clear_array = tom_inf_bat_artillery_terrain_attack_array
		clear_array = tom_inf_bat_artillery_terrain_defense_array
		
		for_each_loop = {
			array = tom_inf_bat_popup_base_stats_array
			value = temp_v
			add_to_array = { tom_inf_bat_artillery_base_stats_array = temp_v }
		}
		for_each_loop = {
			array = tom_inf_bat_popup_combat_stats_array
			value = temp_v
			add_to_array = { tom_inf_bat_artillery_combat_stats_array = temp_v }
		}
		for_each_loop = {
			array = tom_inf_bat_popup_equipment_stats_array_1
			value = temp_v
			add_to_array = { tom_inf_bat_artillery_equipment_stats_array_1 = temp_v }
		}
		for_each_loop = {
			array = tom_inf_bat_popup_equipment_stats_array_2_type
			value = temp_v
			add_to_array = { tom_inf_bat_artillery_equipment_stats_array_2_type = temp_v }
		}
		for_each_loop = {
			array = tom_inf_bat_popup_equipment_stats_array_2_count
			value = temp_v
			add_to_array = { tom_inf_bat_artillery_equipment_stats_array_2_count = temp_v }
		}
		
		for_each_loop = {
			array = tom_inf_bat_popup_terrain_bonus_array
			value = temp_v
			add_to_array = { tom_inf_bat_artillery_terrain_bonus_array = temp_v }
		}
		for_each_loop = {
			array = tom_inf_bat_popup_terrain_movement_array
			value = temp_v
			add_to_array = { tom_inf_bat_artillery_terrain_movement_array = temp_v }
		}
		for_each_loop = {
			array = tom_inf_bat_popup_terrain_attack_array
			value = temp_v
			add_to_array = { tom_inf_bat_artillery_terrain_attack_array = temp_v }
		}
		for_each_loop = {
			array = tom_inf_bat_popup_terrain_defense_array
			value = temp_v
			add_to_array = { tom_inf_bat_artillery_terrain_defense_array = temp_v }
		}
	}
	
	clear_array = tom_inf_battalion_artillery_company_array
	clear_array = tom_inf_battalion_artillery_platoon_array
	
	for_each_loop = {
		array = tom_inf_battalion_company_array
		value = temp_v # Avoid any potential name overlap
		add_to_array = { tom_inf_battalion_artillery_company_array = temp_v }
	}
	for_each_loop = {
		array = tom_inf_battalion_platoon_array
		value = temp_v
		add_to_array = { tom_inf_battalion_artillery_platoon_array = temp_v }
	}
	
	set_variable = { tom_inf_bat_artillery_50_count = tom_inf_bat_popup_50_count }
	set_variable = { tom_inf_bat_artillery_59_count = tom_inf_bat_popup_59_count }
	
	set_variable = { tom_inf_bat_artillery_5000_count = tom_inf_bat_popup_5000_count }
	
	set_variable = { tom_inf_bat_artillery_5900_count = tom_inf_bat_popup_5900_count }

	set_variable = { tom_inf_bat_artillery_5911_count = tom_inf_bat_popup_5911_count }
