tom_inf_supply_depot_build_effect = {
	every_controlled_state = {
		limit = {
			tom_inf_state_bordering_enemy_trigger = yes
			or = {
				supply_node > 0
				naval_base > 0
			}
		}
		
		if = {
			limit = {
				tom_inf_supply_depot < 5
			}
			add_building_construction = {
				type = tom_inf_supply_depot
				level = 1
				instant_build = yes
			}
		}
	}
	
	every_controlled_state = {
		limit = {
			tom_inf_state_bordering_enemy_trigger = no
			tom_inf_supply_depot > 0
		}
		
		remove_building = {
			type = tom_inf_supply_depot
			level = 1
		}
	}
}
