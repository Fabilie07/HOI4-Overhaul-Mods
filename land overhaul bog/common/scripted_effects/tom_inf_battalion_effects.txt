tom_inf_subunit_add_effect = {
	if = {
		limit = {
			check_variable = {
				var = v
				value = -1
				compare = not_equals
			}
		}
		
		if = {
			limit = {
				NOT = {
					has_country_flag = tom_inf_bat_designer_add_company
				}
			}
			
			tom_inf_battalion_platoon_add_effect = yes
			remove_from_array = {
				array = tom_inf_battalion_platoon_array
				index = tom_inf_bat_current_slot
			}
			add_to_array = {
				array = tom_inf_battalion_platoon_array
				value = v
				index = tom_inf_bat_current_slot
			}
		}
		else = {
			tom_inf_battalion_company_add_effect = yes
			remove_from_array = {
				array = tom_inf_battalion_company_array
				index = tom_inf_bat_current_slot
			}
			add_to_array = {
				array = tom_inf_battalion_company_array
				value = v
				index = tom_inf_bat_current_slot
			}
		}
	}
	else = {
		if = {
			limit = {
				NOT = {
					has_country_flag = tom_inf_bat_designer_add_company
				}
			}
			
			tom_inf_battalion_platoon_remove_effect = yes
			remove_from_array = {
				array = tom_inf_battalion_platoon_array
				index = tom_inf_bat_current_slot
			}
			add_to_array = {
				array = tom_inf_battalion_platoon_array
				value = 0
				index = tom_inf_bat_current_slot
			}
		}
		else = {
			tom_inf_battalion_company_remove_effect = yes
			remove_from_array = {
				array = tom_inf_battalion_company_array
				index = tom_inf_bat_current_slot
			}
			add_to_array = {
				array = tom_inf_battalion_company_array
				value = 0
				index = tom_inf_bat_current_slot
			}
		}
	}
}
