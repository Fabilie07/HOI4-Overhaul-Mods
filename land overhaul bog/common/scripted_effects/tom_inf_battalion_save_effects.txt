		}

		limit = {

		}
		
		tom_inf_bat_save_designer_paratrooper_effect = yes
	}
	else_if = {
		limit = {
			has_country_flag = tom_inf_bat_current_marine
		}
		
		tom_inf_bat_save_designer_marine_effect = yes

			has_country_flag = tom_inf_bat_current_marine_commando

		tom_inf_bat_save_designer_marine_commando_effect = yes

		}
		
		tom_inf_bat_save_designer_amphibious_mechanized_effect = yes

		limit = {

		tom_inf_bat_save_designer_rangers_mechanized_effect = yes
	}

	tom_inf_bat_close_designer_effect = yes
