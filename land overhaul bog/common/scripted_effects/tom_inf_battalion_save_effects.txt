tom_inf_compute_equipment_cost_effect = {
	clear_array = tom_inf_bat_needed_equipment_type_array
	clear_array = tom_inf_bat_needed_equipment_count_array
	clear_array = tom_inf_bat_needed_equipment_has_array
	
	for_each_loop = {
		array = tom_inf_bat_popup_equipment_stats_array_2_type
		value = temp_equipment_archetype
		index = temp_equipment_index
		
		set_temp_variable = { temp_equipment_after = tom_inf_bat_popup_equipment_stats_array_2_count^temp_equipment_index }
		set_temp_variable = { temp_equipment_before = tom_inf_bat_popup_equipment_stats_array_2_needed^temp_equipment_index }
		
		if = {
			limit = {
				check_variable = { temp_equipment_after > temp_equipment_before }
			}
			
			subtract_from_temp_variable = { temp_equipment_after = temp_equipment_before }
			if = {
				limit = { has_country_flag = tom_inf_bat_current_infantry }
				multiply_temp_variable = { temp_equipment_after = num_battalions_with_type@infantry }
			}
			else_if = {
				limit = { has_country_flag = tom_inf_bat_current_motorized }
				multiply_temp_variable = { temp_equipment_after = num_battalions_with_type@motorized }
			}
			else_if = {
				limit = { has_country_flag = tom_inf_bat_current_mechanized }
				multiply_temp_variable = { temp_equipment_after = num_battalions_with_type@mechanized }
			}
			else_if = {
				limit = { has_country_flag = tom_inf_bat_current_cavalry }
				multiply_temp_variable = { temp_equipment_after = num_battalions_with_type@cavalry }
			}
			else_if = {
				limit = { has_country_flag = tom_inf_bat_current_mountaineer }
				multiply_temp_variable = { temp_equipment_after = num_battalions_with_type@mountaineers }
			}
			else_if = {
				limit = { has_country_flag = tom_inf_bat_current_paratrooper }
				multiply_temp_variable = { temp_equipment_after = num_battalions_with_type@paratrooper }
			}
			else_if = {
				limit = { has_country_flag = tom_inf_bat_current_marine }
				multiply_temp_variable = { temp_equipment_after = num_battalions_with_type@marine }
			}
			else_if = {
				limit = { has_country_flag = tom_inf_bat_current_marine_commando }
				multiply_temp_variable = { temp_equipment_after = num_battalions_with_type@marine_commando }
			}
			else_if = {
				limit = { has_country_flag = tom_inf_bat_current_ranger }
				multiply_temp_variable = { temp_equipment_after = num_battalions_with_type@ranger_battalion }
			}
			else_if = {
				limit = { has_country_flag = tom_inf_bat_current_mountaineers_mechanized }
				multiply_temp_variable = { temp_equipment_after = num_battalions_with_type@mountaineers_mechanized }
			}
			else_if = {
				limit = { has_country_flag = tom_inf_bat_current_amphibious_mechanized }
				multiply_temp_variable = { temp_equipment_after = num_battalions_with_type@amphibious_mechanized }
			}
			else_if = {
				limit = { has_country_flag = tom_inf_bat_current_rangers_mechanized }
				multiply_temp_variable = { temp_equipment_after = num_battalions_with_type@rangers_mechanized }
			}
			
			tom_inf_compute_temp_equipment_before = yes
			add_to_array = { tom_inf_bat_needed_equipment_type_array = temp_equipment_archetype }
			add_to_array = { tom_inf_bat_needed_equipment_count_array = temp_equipment_after }
			
			tom_inf_compute_temp_equipment_has = yes
			add_to_array = { tom_inf_bat_needed_equipment_has_array = temp_equipment_has }
		}
	}
}
