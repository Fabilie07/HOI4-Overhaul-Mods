on_actions = {
	on_startup = {
		effect = {
			every_country = {		
				set_technology = { tom_inf_indepth_module_package_tech = 1 popup = no }
			}
		}
	}
	
	on_daily = {
		effect = {
			if = {
				limit = {
					NOT = { has_tech = tom_inf_indepth_module_package_tech }
				}
				set_technology = { tom_inf_indepth_module_package_tech = 1 popup = no }
			}
		}
	}
}
