tom_hi_change_clamp_effect = {
	set_temp_variable = { tank_target_current_difference = tom_hi_tank_target }
	subtract_from_temp_variable = { tank_target_current_difference = tom_hi_tank_current }
	if = {
		limit = {
			check_variable = {
				var = tank_target_current_difference
				value = 0
				compare = less_than
			}
		}
		clamp_variable = {
			var = tom_hi_tank_change
			min = tank_target_current_difference
		}
	}
	else = {
		clamp_variable = {
			var = tom_hi_tank_change
			max = tank_target_current_difference
		}
