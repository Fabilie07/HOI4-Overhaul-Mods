nom_sell_ship_dockyards = {
	random_owned_state = {
		limit = {
			NOT = { has_state_flag = nom_state_building_foreign_ship }
			dockyard > 1
		}
		
		set_state_flag = nom_state_building_foreign_ship
		remove_building = {
			type = dockyard
			level = 2
		}
		add_extra_state_shared_building_slots = -2
	}
	add_to_variable = { nom_used_dockyards_commission = 2 }
	add_political_power = 10
}
nom_sell_ship_return_dockyards = {
	random_owned_state = {
		limit = {
			has_state_flag = nom_state_building_foreign_ship
		}
		
		clr_state_flag = nom_state_building_foreign_ship
		add_extra_state_shared_building_slots = 2
		add_building_construction = {
			type = dockyard
			level = 2
			instant_build = yes
		}
	}
	add_to_variable = { nom_used_dockyards_commission = -2 }
	navy_experience = 10
	country_event = NOM_trade.31
}

			limit = { ROOT = { has_country_flag = nom_trade_deal_economy_flag } }

			}
