nom_can_build_foreign_ship = {
	any_owned_state = {
		NOT = { has_state_flag = nom_state_building_foreign_ship }
		dockyard > 1
	}
	country_exists = THIS
	has_capitulated = no

	ROOT = {
		OR = {
			NOT = { has_variable = nom_trade_deal_partner }
			check_variable = { nom_trade_deal_partner = PREV }
		}
		NOT = { is_embargoing = PREV }
		NOT = { tag = PREV }
	}
	NOT = {
		OR = {
			has_war_with = ROOT
			is_embargoing = ROOT
			any_country = {
				is_ally_with = ROOT
				has_war_with = THIS
			}
		}
	}
}

		NOT = { has_state_flag = nom_state_building_foreign_capital_ship }
		dockyard > 3
	}
}
