	desc = NOM_trade.1.desc

	option = {	# Demand economic compensation
		name = NOM_trade.1.a

				has_opinion = { target = FROM value > 30 }
			}
		}
	}

		name = NOM_trade.1.b
		trigger = {
			FROM = {
				OR = {
					steel > 19

					chromium > 9
					rubber > 9
				}
			}
		}
		
		FROM = {
			country_event = {
				id = NOM_trade.3
				hours = 4
			}
		}
		
		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				has_opinion = { target = FROM value < 30 }
			}
			modifier = {
				add = -95
				has_opinion = { target = FROM value > 70 }
			}
		}
	}
	
	option = {	# Accept
		name = NOM_trade.1.c
		
		FROM = {
			country_event = {
				id = NOM_trade.4
				hours = 4
			}
		}
		
		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				has_opinion = { target = FROM value < 70 }
			}
		}
	}
	
	option = {	# Refuse

country_event = {	# FROM withdraws from the Trade Deal
	id = NOM_trade.8
	title = NOM_trade.8.title
	desc = NOM_trade.8.desc
	
	picture = GFX_report_event_generic_sign_treaty1
	
	is_triggered_only = yes
	
	option = {	# Unfortunate
		name = NOM_trade.8.a
		
		remove_from_array = { nom_trade_deal_countries = FROM }

		if = {
			limit = { FROM = { has_country_flag = nom_trade_deal_economy_flag } }
			add_offsite_building = {
				type = industrial_complex
				level = -5
			}
			FROM = { add_to_variable = { nom_trade_compensation_civs = -5 } }
			clr_country_flag = nom_trade_deal_economy_flag
		}
		else_if = {
			limit = { FROM = { has_country_flag = nom_trade_deal_steel_flag } }
			add_to_variable = { nom_trade_compensation_steel = -40 }
			FROM = { add_to_variable = { nom_trade_compensation_steel = 40 } }
			
			custom_override_tooltip = {
				tooltip = {
					localization_key = nom_trade_deal_end_tt
					AMOUNT = "40"
					TYPE = "Steel"
				}
			}
			clr_country_flag = nom_trade_deal_steel_flag
		}
		else_if = {
			limit = { FROM = { has_country_flag = nom_trade_deal_tungsten_flag } }
			add_to_variable = { nom_trade_compensation_tungsten = -20 }
			FROM = { add_to_variable = { nom_trade_compensation_tungsten = 20 } }
			
			custom_override_tooltip = {
				tooltip = {
					localization_key = nom_trade_deal_end_tt
					AMOUNT = "20"
					TYPE = "Tungsten"
				}
			}
			clr_country_flag = nom_trade_deal_tungsten_flag
		}
		else_if = {
			limit = { FROM = { has_country_flag = nom_trade_deal_chromium_flag } }
			add_to_variable = { nom_trade_compensation_chromium = -20 }
			FROM = { add_to_variable = { nom_trade_compensation_chromium = 20 } }
			
			custom_override_tooltip = {
				tooltip = {
					localization_key = nom_trade_deal_end_tt
					AMOUNT = "20"
					TYPE = "Chromium"
				}
			}
			clr_country_flag = nom_trade_deal_chromium_flag
		}
		else_if = {
			limit = { FROM = { has_country_flag = nom_trade_deal_rubber_flag } }
			add_to_variable = { nom_trade_compensation_rubber = -20 }
			FROM = { add_to_variable = { nom_trade_compensation_rubber = 20 } }
			
			custom_override_tooltip = {
				tooltip = {
					localization_key = nom_trade_deal_end_tt
					AMOUNT = "20"
					TYPE = "Rubber"
				}
			}
			clr_country_flag = nom_trade_deal_rubber_flag
		}
		else_if = {
			limit = { FROM = { has_relation_modifier = { target = PREV modifier = nom_trade_deal_signed_modifier } } }
			FROM = { remove_opinion_modifier = { target = PREV modifier = nom_trade_deal_signed_modifier } }
		}
	}
}
