		icon = generic_air
		cost = 100
		
		allowed = {
			NOT = {

					tag = ITA #has a focus
				}
			}
		}
		
		available = {
			OR = {
				AND = {
					OR = {
						has_government = fascism
						has_government = communism
					}
					has_war_support > 0.7
				}
				AND = {
					has_government = neutrality
					has_war_support > 0.8
				}
				AND = {
					has_government = democratic
					has_war_support > 0.9
					has_stability > 0.9
				}
			}
		}
		
		visible = {
			NOT = {
				has_tech = suicide_charge
			}
			has_war = yes
			has_deployed_air_force_size = {
				size > 250
			}
		}
		
		complete_effect = {
			set_rule = {
				can_use_kamikaze_pilots = yes
			}
			set_technology = { suicide_charge = 1 }
			add_stability = -0.05
			add_named_threat = {
				threat = 2
				name = aom_call_for_the_highest_service
			}
		}
	}
